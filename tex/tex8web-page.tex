% --- colors -----------------------------------------------------------------

% \WebColor{name}{light color}{dark color} (xcolor colors) makes the CSS
% variable --name, which is one or the other (see css/tex8web.css)
\newcommand\WebColor[3]{%
    \ifdefined\convertcolorspec
        \web@hex{#2}\web@light
        \web@hex{#3}\web@dark
        \web@Css{:root { --#1: light-dark(\#\web@light, \#\web@dark); }}%
    \else
        \PackageWarning{tex8web}{\string\WebColor\space needs xcolor}%
    \fi
}
\newcommand\web@hex[2]{% color, macro for its hex code
    \extractcolorspecs{#1}\web@model\web@spec
    \convertcolorspec\web@model\web@spec{HTML}#2%
}
% links: hyperref's colors with colorlinks. In a macro, so TeX doesn't meet
% \ifHy@colorlinks when hyperref isn't loaded
\newcommand\web@linkcolors{%
    \ifHy@colorlinks
        \ifdefined\convertcolorspec
            \definecolor{tex8web-link-dark}{RGB}{108, 180, 255}%
            \definecolor{tex8web-cite-dark}{RGB}{95, 211, 160}%
            \definecolor{tex8web-url-dark}{RGB}{242, 127, 176}%
        \fi
        \WebColor{tex8web-link}{\@linkcolor}{tex8web-link-dark}%
        \WebColor{tex8web-cite}{\@citecolor}{tex8web-cite-dark}%
        \WebColor{tex8web-url}{\@urlcolor}{tex8web-url-dark}%
    \fi
}
\@ifpackageloaded{hyperref}{\web@linkcolors}{}

% --- layout -----------------------------------------------------------------

% headings as much larger than the text as in the PDF (in the standard
% classes' sizes), and a column as wide as the PDF's, in ems of the text
\newcommand\web@headsize[2]{% name, size command
    \begingroup#2\xdef\web@size{\f@size pt}\endgroup
    \web@Css{:root { --tex8web-#1-size: \web@em\web@size; }}%
}
\web@headsize{title}{\LARGE}
\web@headsize{section}{\Large}
\web@headsize{subsection}{\large}
\web@headsize{subsubsection}{\normalsize}
\web@Css{:root { --tex8web-textwidth: \web@em\textwidth; }}
