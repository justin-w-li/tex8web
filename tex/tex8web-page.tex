% \WebColor{name}{light color}{dark color} makes the CSS variable --name,

\newcommand\WebColor[3]{%
    \web@hex{#2}\web@light
    \web@hex{#3}\web@dark
    \web@Css{:root { --#1: light-dark(\#\web@light, \#\web@dark); }}%
}
% \web@hex{color}\macro: the color's hex code, which xcolor converts; with
% only the color package (which hyperref's colorlinks load), read from the
% driver's form color keeps (rgb r g b, gray g or cmyk c m y k). The color
% . is the current one
\ExplSyntaxOn
\tl_new:N \l__web_color_tl
\tl_new:N \l__web_hex_tl
\cs_new_protected:Npn \web@hex #1#2 {
    \cs_if_exist:NTF \convertcolorspec
        {
            \extractcolorspecs {#1} \web@model \web@spec
            \convertcolorspec \web@model \web@spec {HTML} #2
        }
        {
            \tl_set:Nn \l__web_hex_tl {000000}
            \str_if_eq:nnTF {#1} {.}
                { \tl_set_eq:NN \l__web_color_tl \current@color }
                { \tl_set:Ne \l__web_color_tl { \cs_if_exist_use:c { \token_to_str:N \color@ #1 } } }
            \tl_if_empty:NF \l__web_color_tl
                {
                    \regex_replace_all:nnN { \s+ } { \  } \l__web_color_tl
                    \exp_after:wN \__web_hex_model:w \l__web_color_tl \q_stop
                }
            \tl_set_eq:NN #2 \l__web_hex_tl
        }
}
\cs_new_protected:Npn \__web_hex_model:w #1 ~ #2 \q_stop {
    \str_case:nn {#1} {
        {rgb} { \__web_hex_rgb:w #2 \q_stop }
        {gray} { \__web_hex_rgb:w #2 ~ #2 ~ #2 \q_stop }
        {cmyk} { \__web_hex_cmyk:w #2 \q_stop }
    }
}
\cs_new_protected:Npn \__web_hex_rgb:w #1 ~ #2 ~ #3 \q_stop {
    \tl_set:Ne \l__web_hex_tl { \__web_hex_byte:n {#1} \__web_hex_byte:n {#2} \__web_hex_byte:n {#3} }
}
\cs_new_protected:Npn \__web_hex_cmyk:w #1 ~ #2 ~ #3 ~ #4 \q_stop {
    \__web_hex_rgb:w {(1-#1)*(1-#4)} ~ {(1-#2)*(1-#4)} ~ {(1-#3)*(1-#4)} \q_stop
}
\cs_new:Npn \__web_hex_byte:n #1 {
    \exp_args:Ne \__web_hex_pad:n { \int_to_Hex:n { \fp_eval:n { round(255*(#1)) } } }
}
\cs_new:Npn \__web_hex_pad:n #1 { \int_compare:nNnT { \tl_count:n {#1} } = 1 {0} #1 }
\ExplSyntaxOff
% \color as a declaration ({\color{red} text}), in a span like \textcolor's
% (html4.4ht), whose \a:color, tex4ht's hook in \color (color.4ht), is
% empty. In a paragraph the span opens at once; from vertical mode, at each
% paragraph until the group ends. A paragraph's end, \par:end, is written
% only before the next paragraph or block, so the span's end goes first in
% it: whichever comes first, the group's end or the paragraph's, closes the
% span. Black is the page's text color, as in tcolorbox's boxes
\ExplSyntaxOn
\tl_new:N \l__web_textcolor_tl
\bool_new:N \l__web_textcolor_bool
\bool_new:N \l__web_owncolor_bool
% (\Tg doesn't leave vertical mode, which \HCode would, starting a paragraph)
\cs_new_protected:Npn \__web_textcolor_end: { \Tg</span> }
\prg_new_conditional:Npnn \__web_textcolor_if_open: { p, T, F } {
    \exp_args:No \tl_if_head_eq_meaning:nNTF { \par:end } \__web_textcolor_end:
        { \prg_return_true: } { \prg_return_false: }
}
\cs_new_protected:Npn \__web_textcolor_open: {
    \HCode{<span~style="color:\l__web_textcolor_tl;">}
    \tl_if_empty:NTF \par:end
        { \cs_gset:Npn \par:end { \__web_textcolor_end: \cs_gset_eq:NN \par:end \empty } }
        { \tl_gput_left:Nn \par:end { \__web_textcolor_end: } }
}
\cs_new_protected:Npn \__web_textcolor_close: {
    \__web_textcolor_end:
    \tl_gset:Ne \par:end { \exp_args:No \tl_tail:n { \par:end } }
}
\cs_new_protected:Npn \__web_textcolor_par: {
    \bool_lazy_and:nnT { \mode_if_horizontal_p: } { ! \tl_if_empty_p:N \l__web_textcolor_tl }
        { \__web_textcolor_if_open:F { \__web_textcolor_open: } }
}
\cs_new_protected:Npn \__web_textcolor_group_end: {
    \__web_textcolor_if_open:T { \__web_textcolor_close: }
    \__web_textcolor_par:
}
% a link's end (\EndLink) closes the spans opened in it, and ends their
% color: hyperref's link text is a group, which tex4ht's (\hyperlink,
% \hyperref, cleveref's) often isn't
\tl_new:N \g__web_textcolor_link_tl
\tl_new:N \g__web_textcolor_linkcolor_tl
\cs_new_protected:Npn \__web_textcolor_link: {
    \tl_gset_eq:NN \g__web_textcolor_link_tl \par:end
    \tl_gset_eq:NN \g__web_textcolor_linkcolor_tl \l__web_textcolor_tl
}
\cs_new_protected:Npn \__web_textcolor_link_end: {
    \bool_until_do:nn
        { \tl_if_eq_p:NN \par:end \g__web_textcolor_link_tl || ! \__web_textcolor_if_open_p: }
        { \__web_textcolor_close: }
    \tl_set_eq:NN \l__web_textcolor_tl \g__web_textcolor_linkcolor_tl
}
% the color is read once it is set (\set@color), as xcolor's
% \color{red!50!black} goes through \a:color twice. A link's color
% (hyperref's \HyColor@UseColor) is the stylesheet's, a \colorbox's (its
% \color@b@x) html4.4ht's, and tcolorbox's own (tcbcol...) its box's, which
% tex8web-boxes.tex sets
\cs_new_protected:Npn \__web_textcolor:n #1 {
    \bool_lazy_or:nnF { \l__web_owncolor_bool } { \str_if_eq_p:ee { \str_range:nnn {#1} {1} {6} } { tcbcol } }
        { \bool_set_true:N \l__web_textcolor_bool }
}
\cs_new_protected:Npn \__web_textcolor_set: {
    \bool_if:NT \l__web_textcolor_bool {
        \bool_set_false:N \l__web_textcolor_bool
        \web@hex{.}\l__web_textcolor_tl
        \str_if_eq:VnTF \l__web_textcolor_tl {000000}
            { \tl_set:Nn \l__web_textcolor_tl { var(--tex8web-text) } }
            { \tl_put_left:Nn \l__web_textcolor_tl {\#} }
        \mode_if_horizontal:T { \__web_textcolor_open: }
        \group_insert_after:N \__web_textcolor_group_end:
    }
}
\cs_new_protected:Npn \__web_owncolor:n #1 {
    \bool_set_true:N \l__web_owncolor_bool
    #1
    \bool_set_false:N \l__web_owncolor_bool
}
\cs_if_exist:NT \a:color {
    \web@needs{par:end}
    \Configure{color}{\__web_textcolor:n}
    \cs_set_eq:NN \__web_set_color: \set@color
    \cs_set:Npn \set@color { \__web_set_color: \__web_textcolor_set: }
    % (not in a paragraph tex4ht writes without its <p>, often empty, before
    % a list or in a float)
    \Configure{@BODY}{
        \cs_set_eq:NN \__web_HtmlPar: \HtmlPar
        \cs_set:Npn \HtmlPar
            { \legacy_if:nTF { :nopar } { \__web_HtmlPar: } { \__web_HtmlPar: \__web_textcolor_par: } }
    }
    \cs_set_eq:NN \__web_Link: \Link
    \cs_set:Npn \Link { \__web_textcolor_link: \__web_Link: }
    \cs_set_eq:NN \__web_EndLink: \EndLink
    \cs_set:Npn \EndLink { \__web_textcolor_link_end: \__web_EndLink: }
    \cs_if_exist:NT \HyColor@UseColor {
        \cs_set_eq:NN \__web_HyColor_UseColor:n \HyColor@UseColor
        \cs_set:Npn \HyColor@UseColor #1 { \__web_owncolor:n { \__web_HyColor_UseColor:n {#1} } }
    }
    \cs_set_eq:NN \__web_color_b_x:nnn \color@b@x
    \cs_set:Npn \color@b@x #1#2#3 {
        \__web_owncolor:n { \__web_color_b_x:nnn {#1} {#2} { \bool_set_false:N \l__web_owncolor_bool #3 } }
    }
    % (none in the page's <title>)
    \Configure{@TITLE}{\bool_set_true:N \l__web_owncolor_bool}
    % (article.4ht writes the title's, author's and date's elements in the
    % group of their text: a \color in it gets a group of its own)
    \clist_map_inline:nn { ttl, author, date } {
        \cs_if_exist:cT { a:#1 } {
            \tl_gput_right:cn { a:#1 } { \begingroup }
            \tl_gput_left:cn { b:#1 } { \endgroup }
        }
    }
    % a footnote is in the text's color in the PDF, which LaTeX's output
    % routine gives it
    \clist_map_inline:nn { \@footnotetext, \@mpfootnotetext } {
        \cs_set_eq:cN { __web_ \cs_to_str:N #1 :n } #1
        \cs_set:Npn #1 ##1 {
            \group_begin:
                \tl_clear:N \l__web_textcolor_tl
                \use:c { __web_ \cs_to_str:N #1 :n } {##1}
            \group_end:
        }
    }
}

% headings as much larger than the text as in the PDF (in the standard
% classes' sizes), and a column as wide as the PDF's, in ems of the text.
% With titlesec, a heading's size, family, weight, style and alignment are
% those of its \titleformat (tex8web-hooks.sty), tried out in a box that's
% thrown away, and the space above and below it that of its \titlespacing
% (or the class's, which titlesec reads when it loads). A number as far
% from its title as in the PDF: titlesec's space after the label, KOMA's
% (the glue its \sectionformat ends with), amsart's \enspace (which its
% \@sect puts there, not \@seccntformat), or LaTeX's \quad
% (css/tex8web.css)
\newcommand\web@headsize[2]{% name, size command
    \begingroup
        \def\web@font{#2}%
        \ifcsname web@titlefont@#1\endcsname
            \expandafter\let\expandafter\web@font\csname web@titlefont@#1\endcsname
        \fi
        \global\let\web@spacing\relax
        \global\let\web@labelsep\relax
        \ifcsname ttls@#1\endcsname
            \expandafter\expandafter\expandafter\web@titlespacing\csname ttls@#1\endcsname
        \fi
        \setbox0\vbox{\web@font
            \ifx\web@spacing\relax\else
                \xdef\web@spacing{--tex8web-#1-before: \web@em\web@before;
                    --tex8web-#1-after: \web@em\web@after;
                    --tex8web-#1-after-text: \web@aftertext;}%
            \fi
            \ifcsname c@#1\endcsname
                \ifcsname web@titlesep@#1\endcsname
                    \setlength\dimen@{\csname web@titlesep@#1\endcsname}%
                    \xdef\web@labelsep{\web@em\dimen@}%
                \else\ifcsname #1format\endcsname
                    \setbox0\hbox{\csname #1format\endcsname}\dimen@\wd0
                    \setbox0\hbox{\csname #1format\endcsname\unskip}\advance\dimen@-\wd0
                    \xdef\web@labelsep{\web@em\dimen@}%
                \else\ifdefined\@secnumfont
                    \gdef\web@labelsep{0.5em}%
                \fi\fi\fi
            \fi
            \xdef\web@size{\f@size pt}%
            \gdef\web@family{inherit}%
            \web@iffont\f@family\sfdefault{\gdef\web@family{var(--tex8web-sans)}}%
            \web@iffont\f@family\ttdefault{\gdef\web@family{var(--tex8web-mono)}}%
            \xdef\web@weight{\if b\expandafter\@car\f@series\@nil bold\else normal\fi}%
            \gdef\web@style{normal}%
            \web@iffont\f@shape\itdefault{\gdef\web@style{italic}}%
            \web@iffont\f@shape\sldefault{\gdef\web@style{italic}}%
            \gdef\web@variant{normal}%
            \web@iffont\f@shape\scdefault{\gdef\web@variant{small-caps}}}%
        % (by the commands in the format: tex4ht's \centering leaves the
        % line's skips alone, latex.4ht)
        \gdef\web@align{left}%
        \web@formathas\center{\gdef\web@align{center}}%
        \web@formathas\centering{\gdef\web@align{center}}%
        \web@formathas\Centering{\gdef\web@align{center}}%
        \web@formathas\filcenter{\gdef\web@align{center}}%
        \web@formathas\raggedleft{\gdef\web@align{right}}%
        \web@formathas\RaggedLeft{\gdef\web@align{right}}%
        \web@formathas\filleft{\gdef\web@align{right}}%
    \endgroup
    \web@Css{:root { --tex8web-#1-size: \web@em\web@size; }}%
    \ifx\web@spacing\relax\else\web@Css{:root { \web@spacing }}\fi
    \ifx\web@labelsep\relax\else\web@Css{:root { --tex8web-#1-labelsep: \web@labelsep; }}\fi
    \ifcsname web@titlefont@#1\endcsname
        \web@Css{:root { --tex8web-#1-family: \web@family;
            --tex8web-#1-weight: \web@weight; --tex8web-#1-style: \web@style;
            --tex8web-#1-variant: \web@variant; --tex8web-#1-align: \web@align; }}%
    \fi
}
% a \paragraph or \subparagraph that titlesec gives any shape but runin, on
% a line of its own
\newcommand\web@headdisplay[1]{%
    \ifcsname web@titleshape@#1\endcsname
        \edef\web@a{\csname web@titleshape@#1\endcsname}\def\web@b{runin}%
        \ifx\web@a\web@b\else\web@Css{:root { --tex8web-#1-display: block; }}\fi
    \fi
}
% titlesec's {left}{right}{before}{after}{afterindent}: before and after
% as \web@before and \web@after, in the text's font, as titlesec sets them,
% and after in the text's ems, as \web@aftertext
% (\ttl@assign reads its *n form). (Not left in \@tempskipa: a format's
% \center sets it)
\def\web@titlespacing#1#2#3#4#5{%
    \ttl@assign\@tempskipa#3\relax\beforetitleunit
    \edef\web@before{\the\dimexpr\@tempskipa}%
    \ttl@assign\@tempskipa#4\relax\aftertitleunit
    \edef\web@after{\the\dimexpr\@tempskipa}%
    \edef\web@aftertext{\web@em\web@after}%
    \global\let\web@spacing\@empty}
