% \WebColor{name}{light color}{dark color} makes the CSS variable --name,
% which is one or the other (see css/tex8web.css)
\newcommand\WebColor[3]{%
    \web@hex{#2}\web@light
    \web@hex{#3}\web@dark
    \web@Css{:root { --#1: light-dark(\#\web@light, \#\web@dark); }}%
}
% \web@hex{color}\macro: the color's hex code, which xcolor converts; with
% only the color package (which hyperref's colorlinks load), read from the
% driver's form color keeps (rgb r g b, gray g or cmyk c m y k)
\ExplSyntaxOn
\tl_new:N \l__web_color_tl
\tl_new:N \l__web_hex_tl
\cs_new_protected:Npn \web@hex #1#2 {
    \cs_if_exist:NTF \convertcolorspec
        {
            \extractcolorspecs {#1} \web@model \web@spec
            \convertcolorspec \web@model \web@spec {HTML} #2
        }
        {
            \tl_set:Nn \l__web_hex_tl {000000}
            \cs_if_exist:cT { \token_to_str:N \color@ #1 }
                {
                    \tl_set:Ne \l__web_color_tl { \use:c { \token_to_str:N \color@ #1 } }
                    \regex_replace_all:nnN { \s+ } { \  } \l__web_color_tl
                    \exp_after:wN \__web_hex_model:w \l__web_color_tl \q_stop
                }
            \tl_set_eq:NN #2 \l__web_hex_tl
        }
}
\cs_new_protected:Npn \__web_hex_model:w #1 ~ #2 \q_stop {
    \str_case:nn {#1} {
        {rgb} { \__web_hex_rgb:w #2 \q_stop }
        {gray} { \__web_hex_rgb:w #2 ~ #2 ~ #2 \q_stop }
        {cmyk} { \__web_hex_cmyk:w #2 \q_stop }
    }
}
\cs_new_protected:Npn \__web_hex_rgb:w #1 ~ #2 ~ #3 \q_stop {
    \tl_set:Ne \l__web_hex_tl { \__web_hex_byte:n {#1} \__web_hex_byte:n {#2} \__web_hex_byte:n {#3} }
}
\cs_new_protected:Npn \__web_hex_cmyk:w #1 ~ #2 ~ #3 ~ #4 \q_stop {
    \__web_hex_rgb:w {(1-#1)*(1-#4)} ~ {(1-#2)*(1-#4)} ~ {(1-#3)*(1-#4)} \q_stop
}
\cs_new:Npn \__web_hex_byte:n #1 {
    \exp_args:Ne \__web_hex_pad:n { \int_to_Hex:n { \fp_eval:n { round(255*(#1)) } } }
}
\cs_new:Npn \__web_hex_pad:n #1 { \int_compare:nNnT { \tl_count:n {#1} } = 1 {0} #1 }
\ExplSyntaxOff
% links: hyperref's colors with colorlinks; otherwise the PDF draws boxes,
% and the links keep the color of the text. tex4ht turns colorlinks on
% itself, so count it as the document's when its hyperref options name it,
% or its link colors aren't hyperref's defaults
\ExplSyntaxOn
\bool_new:N \l__web_colorlinks_bool
\cs_new_protected:Npn \web@linkcolors {
    \bool_set:Nn \l__web_colorlinks_bool
        { ! \str_if_eq_p:ee { \@linkcolor / \@citecolor / \@urlcolor } { red / green / magenta } }
    \cs_if_exist:cT { @raw@opt@hyperref.sty }
        { \tl_if_in:cnT { @raw@opt@hyperref.sty } { colorlinks } { \bool_set_true:N \l__web_colorlinks_bool } }
    \bool_lazy_and:nnT { \legacy_if_p:n { Hy@colorlinks } } { \l__web_colorlinks_bool }
        {
            \cs_if_exist:NTF \convertcolorspec
                {
                    \definecolor{tex8web-link-dark}{RGB}{108,180,255}
                    \definecolor{tex8web-cite-dark}{RGB}{95,211,160}
                    \definecolor{tex8web-url-dark}{RGB}{242,127,176}
                }
                {
                    \definecolor{tex8web-link-dark}{rgb}{0.4235,0.7059,1}
                    \definecolor{tex8web-cite-dark}{rgb}{0.3725,0.8275,0.6275}
                    \definecolor{tex8web-url-dark}{rgb}{0.949,0.498,0.6902}
                }
            \WebColor{tex8web-link}{\@linkcolor}{tex8web-link-dark}
            \WebColor{tex8web-cite}{\@citecolor}{tex8web-cite-dark}
            \WebColor{tex8web-url}{\@urlcolor}{tex8web-url-dark}
        }
}

% headings as much larger than the text as in the PDF (in the standard
% classes' sizes), and a column as wide as the PDF's, in ems of the text.
% With titlesec, a heading's size, family, weight, style and alignment are
% those of its \titleformat (tex8web-hooks.sty), tried out in a box that's
% thrown away, and the space above and below it that of its \titlespacing
% (or the class's, which titlesec reads when it loads). A number as far
% from its title as in the PDF: titlesec's space after the label, KOMA's
% (the glue its \sectionformat ends with), amsart's \enspace (which its
% \@sect puts there, not \@seccntformat), or LaTeX's \quad
% (css/tex8web.css)
\newcommand\web@headsize[2]{% name, size command
    \begingroup
        \def\web@font{#2}%
        \ifcsname web@titlefont@#1\endcsname
            \expandafter\let\expandafter\web@font\csname web@titlefont@#1\endcsname
        \fi
        \global\let\web@spacing\relax
        \global\let\web@labelsep\relax
        \ifcsname ttls@#1\endcsname
            \expandafter\expandafter\expandafter\web@titlespacing\csname ttls@#1\endcsname
        \fi
        \setbox0\vbox{\web@font
            \ifx\web@spacing\relax\else
                \xdef\web@spacing{--tex8web-#1-before: \web@em\web@before;
                    --tex8web-#1-after: \web@em\web@after;}%
            \fi
            \ifcsname c@#1\endcsname
                \ifcsname web@titlesep@#1\endcsname
                    \setlength\dimen@{\csname web@titlesep@#1\endcsname}%
                    \xdef\web@labelsep{\web@em\dimen@}%
                \else\ifcsname #1format\endcsname
                    \setbox0\hbox{\csname #1format\endcsname}\dimen@\wd0
                    \setbox0\hbox{\csname #1format\endcsname\unskip}\advance\dimen@-\wd0
                    \xdef\web@labelsep{\web@em\dimen@}%
                \else\ifdefined\@secnumfont
                    \gdef\web@labelsep{0.5em}%
                \fi\fi\fi
            \fi
            \xdef\web@size{\f@size pt}%
            \gdef\web@family{inherit}%
            \web@iffont\f@family\sfdefault{\gdef\web@family{var(--tex8web-sans)}}%
            \web@iffont\f@family\ttdefault{\gdef\web@family{var(--tex8web-mono)}}%
            \xdef\web@weight{\if b\expandafter\@car\f@series\@nil bold\else normal\fi}%
            \gdef\web@style{normal}%
            \web@iffont\f@shape\itdefault{\gdef\web@style{italic}}%
            \web@iffont\f@shape\sldefault{\gdef\web@style{italic}}%
            \gdef\web@variant{normal}%
            \web@iffont\f@shape\scdefault{\gdef\web@variant{small-caps}}}%
        % (by the commands in the format: tex4ht's \centering leaves the
        % line's skips alone, latex.4ht)
        \gdef\web@align{left}%
        \web@formathas\center{\gdef\web@align{center}}%
        \web@formathas\centering{\gdef\web@align{center}}%
        \web@formathas\Centering{\gdef\web@align{center}}%
        \web@formathas\filcenter{\gdef\web@align{center}}%
        \web@formathas\raggedleft{\gdef\web@align{right}}%
        \web@formathas\RaggedLeft{\gdef\web@align{right}}%
        \web@formathas\filleft{\gdef\web@align{right}}%
    \endgroup
    \web@Css{:root { --tex8web-#1-size: \web@em\web@size; }}%
    \ifx\web@spacing\relax\else\web@Css{:root { \web@spacing }}\fi
    \ifx\web@labelsep\relax\else\web@Css{:root { --tex8web-#1-labelsep: \web@labelsep; }}\fi
    \ifcsname web@titlefont@#1\endcsname
        \web@Css{:root { --tex8web-#1-family: \web@family;
            --tex8web-#1-weight: \web@weight; --tex8web-#1-style: \web@style;
            --tex8web-#1-variant: \web@variant; --tex8web-#1-align: \web@align; }}%
    \fi
}
% a \paragraph or \subparagraph that titlesec gives any shape but runin, on
% a line of its own
\newcommand\web@headdisplay[1]{%
    \ifcsname web@titleshape@#1\endcsname
        \edef\web@a{\csname web@titleshape@#1\endcsname}\def\web@b{runin}%
        \ifx\web@a\web@b\else\web@Css{:root { --tex8web-#1-display: block; }}\fi
    \fi
}
% titlesec's {left}{right}{before}{after}{afterindent}: before and after
% as \web@before and \web@after, in the text's font, as titlesec sets them
% (\ttl@assign reads its *n form). (Not left in \@tempskipa: a format's
% \center sets it)
\def\web@titlespacing#1#2#3#4#5{%
    \ttl@assign\@tempskipa#3\relax\beforetitleunit
    \edef\web@before{\the\dimexpr\@tempskipa}%
    \ttl@assign\@tempskipa#4\relax\aftertitleunit
    \edef\web@after{\the\dimexpr\@tempskipa}%
    \global\let\web@spacing\@empty}
% #2 if the heading's format (\web@font) holds the command #1
\newcommand\web@formathas[2]{%
    \def\web@a{\in@#1}\expandafter\web@a\expandafter{\web@font}\ifin@#2\fi}
% #3 if font macros #1 and #2 name the same font (\ifx alone would also
% compare whether they're \long)
\newcommand\web@iffont[3]{%
    \edef\web@a{#1}\edef\web@b{#2}\ifx\web@a\web@b#3\fi}
% (and the title in the size its text starts with, \title{\Large ..}, as
% in \maketitle's {\LARGE \@title})
\long\def\web@titlefirst#1#2\@nil{%
    \in@{#1}{\tiny\scriptsize\footnotesize\small\normalsize\large\Large\LARGE\huge\Huge}%
    \ifin@\def\web@titlesize{#1}\else\let\web@titlesize\@empty\fi}
\ifdefined\@title\expandafter\web@titlefirst\@title\@empty\@nil\else\let\web@titlesize\@empty\fi
\web@headsize{title}{\LARGE\web@titlesize}
\web@headsize{section}{\Large}
\web@headsize{subsection}{\large}
\web@headsize{subsubsection}{\normalsize}
\web@headsize{paragraph}{\normalsize}
\web@headsize{subparagraph}{\normalsize}
\web@headdisplay{paragraph}
\web@headdisplay{subparagraph}
\web@Css{:root { --tex8web-textwidth: \web@em\textwidth; }}
