% \WebColor{name}{light color}{dark color}: the CSS variable --name
\newcommand\WebColor[3]{%
    \web@hex{#2}\web@light
    \web@hex{#3}\web@dark
    \web@Css{:root { --#1: light-dark(\#\web@light, \#\web@dark); }}%
}
% \web@hex{color}\macro: the color's hex code, converted by xcolor. With
% only the color package (loaded by hyperref's colorlinks), the code is read
% from the driver's form that color keeps: rgb r g b, gray g, or
% cmyk c m y k. The color . is the current one
\ExplSyntaxOn
\tl_new:N \l__web_color_tl
\tl_new:N \l__web_hex_tl
\cs_new_protected:Npn \web@hex #1#2 {
    \cs_if_exist:NTF \convertcolorspec
        {
            \extractcolorspecs {#1} \web@model \web@spec
            \convertcolorspec \web@model \web@spec {HTML} #2
        }
        {
            \tl_set:Nn \l__web_hex_tl {000000}
            \str_if_eq:nnTF {#1} {.}
                { \tl_set_eq:NN \l__web_color_tl \current@color }
                { \tl_set:Ne \l__web_color_tl { \cs_if_exist_use:c { \token_to_str:N \color@ #1 } } }
            \tl_if_empty:NF \l__web_color_tl
                {
                    \regex_replace_all:nnN { \s+ } { \  } \l__web_color_tl
                    \exp_after:wN \__web_hex_model:w \l__web_color_tl \q_stop
                }
            \tl_set_eq:NN #2 \l__web_hex_tl
        }
}
\cs_new_protected:Npn \__web_hex_model:w #1 ~ #2 \q_stop {
    \str_case:nn {#1} {
        {rgb} { \__web_hex_rgb:w #2 \q_stop }
        {gray} { \__web_hex_rgb:w #2 ~ #2 ~ #2 \q_stop }
        {cmyk} { \__web_hex_cmyk:w #2 \q_stop }
    }
}
\cs_new_protected:Npn \__web_hex_rgb:w #1 ~ #2 ~ #3 \q_stop {
    \tl_set:Ne \l__web_hex_tl { \__web_hex_byte:n {#1} \__web_hex_byte:n {#2} \__web_hex_byte:n {#3} }
}
\cs_new_protected:Npn \__web_hex_cmyk:w #1 ~ #2 ~ #3 ~ #4 \q_stop {
    \__web_hex_rgb:w {(1-#1)*(1-#4)} ~ {(1-#2)*(1-#4)} ~ {(1-#3)*(1-#4)} \q_stop
}
\cs_new:Npn \__web_hex_byte:n #1 {
    \exp_args:Ne \__web_hex_pad:n { \int_to_Hex:n { \fp_eval:n { round(255*(#1)) } } }
}
\cs_new:Npn \__web_hex_pad:n #1 { \int_compare:nNnT { \tl_count:n {#1} } = 1 {0} #1 }
% \color as a declaration ({\color{red} text}), which html4.4ht drops: its
% \a:color, tex4ht's hook in \color, is empty. tex8web writes a mark where
% the color is set and one with the same id where its group ends, and
% text.lua's colors turns the pair into a span. The color is read in
% \set@color, because xcolor's \color{red!50!black} calls \a:color twice.
% Colors that are not the text's get no mark: tcolorbox's (tcbcol...) are
% set on the box by tex8web-boxes.tex, a link's comes from the stylesheet
% (hyperref's \HyColor@UseColor), and a \colorbox's background is written
% by html4.4ht (color.4ht's \color@b@x)
\bool_new:N \l__web_declared_bool
\bool_new:N \l__web_own_bool
\int_new:N \g__web_declared_int
\tl_new:N \l__web_declared_tl
\cs_new_protected:Npn \__web_declared:n #1 {
    \bool_lazy_or:nnF { \l__web_own_bool } { \str_if_eq_p:ee { \str_range:nnn {#1} {1} {6} } { tcbcol } }
        { \bool_set_true:N \l__web_declared_bool }
}
\cs_new_protected:Npn \__web_own:n #1 {
    \bool_set_true:N \l__web_own_bool
    #1
    \bool_set_false:N \l__web_own_bool
}
\cs_new_protected:Npn \__web_declared_mark: {
    \bool_if:NT \l__web_declared_bool {
        \bool_set_false:N \l__web_declared_bool
        \web@hex{.}\l__web_declared_tl
        \int_gincr:N \g__web_declared_int
        % (\Tg, because \HCode would leave vertical mode)
        \Tg<span~class="tex8web-color"~data-id="\int_use:N \g__web_declared_int"
            ~data-color="\#\l__web_declared_tl"/>
        \cs_gset:cpe { __web_declared_ \int_use:N \g__web_declared_int : } {
            \exp_not:N \Tg <span~class="tex8web-color-end"~data-id="\int_use:N \g__web_declared_int"/>
        }
        \exp_args:Nc \group_insert_after:N { __web_declared_ \int_use:N \g__web_declared_int : }
    }
}

% colortbl's \cellcolor, \rowcolor, and \columncolor. tex4ht converts the
% color for the cell's CSS rule itself (colortbl.4ht, html4.4ht's
% \add:color): it writes a wrong rgb() for a name that xcolor or dvips has
% and CSS lacks, and truncates other colors. Dark mode needs the hex colors
% the rest of the page has, so a name or expression that xcolor knows goes
% through \web@hex. Colors tex4ht makes, such as tblcol-1 or a color with a
% model, stay as they are.
\@ifpackageloaded{colortbl}{\web@needs{add:color}}{}

    {
        \cs_set_eq:NN \__web_add_color:nnn \add:color

            {
                \tl_set:Ne \l__web_cell_tl {#2}
                \bool_if:NTF \g__web_xcolor_bool
                    {
                        \tl_if_in:NnTF \l__web_cell_tl { ! }
                            { \__web_cell_color:nVn {#1} \l__web_cell_tl {#3} }
                            {
                                \tl_if_head_eq_charcode:VNTF \l__web_cell_tl -
                                    { \__web_cell_color:nVn {#1} \l__web_cell_tl {#3} }
                                    {
                                        \exp_args:NV \@ifundefinedcolor \l__web_cell_tl
                                            { \__web_add_color:nnn {#1} {#2} {#3} }
                                            { \__web_cell_color:nVn {#1} \l__web_cell_tl {#3} }
                                    }
                            }
                    }
                    { \__web_add_color:nnn {#1} {#2} {#3} }
            }
        \cs_new_protected:Npn \__web_cell_color:nnn #1 #2 #3
            {
                \web@hex {#2} \l__web_hex_tl
                \web@Css { \# TBL- \TableNo - #3 { ~ #1 : \# \l__web_hex_tl } }
            }
        \cs_generate_variant:Nn \__web_cell_color:nnn { nVn }
        \cs_if_exist:NT \rowcolors { \web@needs{a:rowcolors} }
        \cs_if_exist:NT \a:rowcolors
            {
                \web@needs{rowc@l@rs}
                \web@needs{@rowc@lors}
                \web@needs{halignTR}
                \web@needs{o:noalign:}
                % \rowcolors colors a row from array's row hooks, which run in
                % array's \halign, as old array's \everycr did (xcolor.4ht
                % relies on \everycr). tex4ht builds another \halign, so the
                % color is set as tex4ht starts a row, in \halignTR; html4.4ht's
                % \Configure{rowcolors} never draws it. The color goes on the
                % row, under a \rowcolor or \cellcolor on its cells, as in the
                % PDF. \rowcolors' commands for each row, such as [\hline], are
                % dropped, because they cannot run there.
                \cs_new_protected:Npn \__web_row_color:n #1
                    { \add:color { background-color } {#1} { \HRow - } }

                \cs_set_eq:NN \__web_rowcolors:w \rowc@l@rs
                \cs_set:Npn \rowc@l@rs [ #1 ] { \__web_rowcolors:w [ ] }
                \cs_new_protected:Npn \__web_row_colors:
                    {
                        \group_begin:
                            \cs_set_eq:NN \o:noalign: \use:n
                            \@rowc@lors
                        \group_end:
                    }
                \append:def \halignTR { \__web_row_colors: }
            }
    }

% heading sizes relative to the text, as in the PDF (the standard classes'
% sizes), and a column as wide as the PDF's, in ems of the text. A
% heading's size, family, weight, style, and alignment come from its
% titlesec \titleformat (tex8web-hooks.sty), else from the class (its
% \@startsection style, or memoir's chapter and part fonts), tried in a box
% that is thrown away. \web@headfont@<name> is defined when a font is read
% from titlesec or the class. Its space above and below comes from its
% \titlespacing, or from the class, which titlesec reads when it loads. The
% space between number and title, as in the PDF: titlesec's space after
% the label, KOMA's (the glue its \sectionformat ends with), amsart's
% \enspace (put there by its \@sect, not \@seccntformat), or LaTeX's \quad
% (css/tex8web.css)
\newcommand\web@headsize[2]{% name, size command without titlesec or the class's font
    \begingroup
        \def\web@font{#2}%
        \global\let\web@spacing\relax
        \global\let\web@labelsep\relax
        \ifcsname ttls@#1\endcsname
            \web@needs{ttl@assign}%
            \expandafter\expandafter\expandafter\web@titlespacing\csname ttls@#1\endcsname
        \else
            \web@startsection{#1}%
        \fi
        \ifcsname web@titlefont@#1\endcsname
            \expandafter\let\expandafter\web@classfont\csname web@titlefont@#1\endcsname
        \fi
        \ifdefined\web@classfont
            \let\web@font\web@classfont
            \expandafter\global\expandafter\let\csname web@headfont@#1\endcsname\@empty
        \fi
        % (each font is given an empty group, for a font whose last command
        % takes the title, as memoir's part and chapter fonts may:
        % \MakeUppercase)
        \setbox0\vbox{\web@font{}%
            \ifx\web@spacing\relax\else
                \xdef\web@spacing{--tex8web-#1-before: \web@em\web@before;
                    --tex8web-#1-after: \web@em\web@after;
                    --tex8web-#1-after-text: \web@aftertext;}%
            \fi
            \ifcsname c@#1\endcsname
                \ifcsname web@titlesep@#1\endcsname
                    \setlength\dimen@{\csname web@titlesep@#1\endcsname}%
                    \xdef\web@labelsep{\web@em\dimen@}%
                \else\ifcsname #1format\endcsname
                    \setbox0\hbox{\csname #1format\endcsname}\dimen@\wd0
                    \setbox0\hbox{\csname #1format\endcsname\unskip}\advance\dimen@-\wd0
                    \xdef\web@labelsep{\web@em\dimen@}%
                \else\ifdefined\@secnumfont
                    \gdef\web@labelsep{0.5em}%
                \fi\fi\fi
            \fi
            \web@readfont}%
        % (alignment read from the commands in the format, because tex4ht's
        % \centering leaves the line's skips alone, latex.4ht)
        \gdef\web@align{left}%
        \web@formathas\center{\gdef\web@align{center}}%
        \web@formathas\centering{\gdef\web@align{center}}%
        \web@formathas\Centering{\gdef\web@align{center}}%
        \web@formathas\filcenter{\gdef\web@align{center}}%
        \web@formathas\raggedleft{\gdef\web@align{right}}%
        \web@formathas\RaggedLeft{\gdef\web@align{right}}%
        \web@formathas\filleft{\gdef\web@align{right}}%
    \endgroup
    \web@Css{:root { --tex8web-#1-size: \web@em\web@size; }}%
    \ifx\web@spacing\relax\else\web@Css{:root { \web@spacing }}\fi
    \ifx\web@labelsep\relax\else\web@Css{:root { --tex8web-#1-labelsep: \web@labelsep; }}\fi
    \ifcsname web@headfont@#1\endcsname
        \web@Css{:root { --tex8web-#1-family: \web@family;
            --tex8web-#1-weight: \web@weight; --tex8web-#1-style: \web@style;
            --tex8web-#1-variant: \web@variant; --tex8web-#1-align: \web@align; }}%
    \fi
}
% a \paragraph or \subparagraph with a titlesec shape other than runin is

\newcommand\web@headdisplay[1]{%
    \ifcsname web@titleshape@#1\endcsname
        \edef\web@a{\csname web@titleshape@#1\endcsname}\def\web@b{runin}%
        \ifx\web@a\web@b\else\web@Css{:root { --tex8web-#1-display: block; }}\fi
    \fi
}
% titlesec's {left}{right}{before}{after}{afterindent}: before and after
% in \web@before and \web@after, read in the text's font as titlesec reads
% them (\ttl@assign handles the *n form), and after in the text's ems in
% \web@aftertext. Not kept in \@tempskipa, which a format's \center sets
\def\web@titlespacing#1#2#3#4#5{%
    \ttl@assign\@tempskipa#3\relax\beforetitleunit
    \edef\web@before{\the\dimexpr\@tempskipa}%
    \ttl@assign\@tempskipa#4\relax\aftertitleunit
    \edef\web@after{\the\dimexpr\@tempskipa}%
    \edef\web@aftertext{\web@em\web@after}%
    \global\let\web@spacing\@empty}
