% tex8web: the tex4ht configuration for HTML that looks like the PDF, with
% the math left as LaTeX source for MathJax. A config \input's this file
% after \begin{document} (see tex8web.cfg). tex8web.mk4 puts tex/ on TeX's
% search path.
\makeatletter

% babel may have made some punctuation active by now: " in German, and
% : ; ! ? in French. The modules are read with these characters ordinary,
% and the catcodes are restored after. Until the call below, this file
% avoids these characters, = in particular.
\def\web@restorecatcodes{}
\def\web@othercatcodes#1,{%
    \ifx\relax#1\else
        \edef\web@restorecatcodes{\web@restorecatcodes
            \catcode#1 \the\catcode#1\relax}%
        \catcode#1 12
        \expandafter\web@othercatcodes
    \fi
}
\web@othercatcodes 33,34,39,58,59,60,61,62,63,96,\relax,

% a length as a multiple of another length, of the body font size, and in ems
\newcommand\web@over[2]{\strip@pt\dimexpr#1*65536/\dimexpr#2\relax\relax}
\newcommand\web@ratio[1]{\web@over{#1}{\f@size pt}}
\newcommand\web@em[1]{\web@ratio{#1}em}
% the text width as the document begins (the config runs after every
% \begin{document} hook, geometry's included)
\newdimen\web@textwidth
\web@textwidth\textwidth
% \Css, with its argument expanded now rather than when tex4ht writes it
\newcommand\web@Css[1]{\edef\web@css{\noexpand\Css{#1}}\web@css}
% the place of the static styles: lua/tex8web/assets.lua puts
% css/tex8web.css here, after tex4ht's rules and before a config's
\Css{/* tex8web.css */}
\makeatother

% tex4ht's split options (1 to 7, or splitat= with a level name) write one
% page per section. tex8web builds one page per document.
\ExplSyntaxOn
\msg_new:nnn { tex8web } { split }
    { tex8web~makes~one~page~per~document;~remove~tex4ht's~split~option~(a~number~or~splitat=~in~\token_to_str:N \Preamble) }
\clist_map_inline:nn { 1, 2, 3, 4, 5, 6, 7 }
    { \:CheckOption {#1} \if:Option \msg_fatal:nn { tex8web } { split } \fi }
\ExplSyntaxOff

\input{tex8web-math.tex}
\input{tex8web-text.tex}
\input{tex8web-page.tex}
\input{tex8web-boxes.tex}

\makeatletter
\web@restorecatcodes
\makeatother
