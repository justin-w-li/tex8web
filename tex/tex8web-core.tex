% tex8web: tex4ht configuration for HTML that looks like the PDF, with MathJax
% reading the math's LaTeX source. A tex4ht config \input's this after
% \begin{document} (see tex8web.cfg); tex8web.mk4 puts this directory on TeX's
% search path.
\makeatletter

% babel may have made some punctuation active by now (" in German, : ; ! ?
% in French); read the modules with these characters ordinary, and restore
% them after. (This file avoids them, and = in particular)
\def\web@restorecatcodes{}
\def\web@othercatcodes#1,{%
    \ifx\relax#1\else
        \edef\web@restorecatcodes{\web@restorecatcodes
            \catcode#1 \the\catcode#1\relax}%
        \catcode#1 12
        \expandafter\web@othercatcodes
    \fi
}
\web@othercatcodes 33,34,39,58,59,60,61,62,63,96,\relax,

% a length as a multiple of the body font size, and in ems of it
\newcommand\web@ratio[1]{\strip@pt\dimexpr#1*65536/\dimexpr\f@size pt\relax\relax}
\newcommand\web@em[1]{\web@ratio{#1}em}
% the text width as the document begins (the config runs after every
% \begin{document} hook, geometry's included)
\newdimen\web@textwidth
\web@textwidth\textwidth
% \Css, with its argument expanded now rather than when tex4ht writes it
\newcommand\web@Css[1]{\edef\web@css{\noexpand\Css{#1}}\web@css}
% the place of the static styles: tex8web.mk4 puts css/tex8web.css here,
% after tex4ht's own rules and before a config's
\Css{/* tex8web.css */}
\makeatother

% tex4ht's split options (1 to 7, or splitat= with a level name) write a
% page per section, and the build makes one page
\ExplSyntaxOn
\msg_new:nnn { tex8web } { split }
    { tex8web~makes~one~page~per~document;~remove~tex4ht's~split~option~(a~number~or~splitat=~in~\token_to_str:N \Preamble) }
\clist_map_inline:nn { 1, 2, 3, 4, 5, 6, 7 }
    { \:CheckOption {#1} \if:Option \msg_fatal:nn { tex8web } { split } \fi }
\ExplSyntaxOff

\input{tex8web-math.tex}
\input{tex8web-text.tex}
\input{tex8web-page.tex}
\input{tex8web-boxes.tex}

\makeatletter
\web@restorecatcodes
\makeatother
